Drawing Software

The purpose of my drawing software was to create a drawing tool that could make digital art reminiscent of memes online like wojaks and such. To accomplish this, I created brushes that made basic shapes and lines in varying color and line quality.

let tool = 1; function setup() { createCanvas(800, 800); background(255); } function draw() { if (mouseIsPressed) { if (tool === 1) { // Regular stroke(0); line(mouseX, mouseY, pmouseX, pmouseY); } else if (tool === 2) { // Blocks noStroke(); fill(0); rect(mouseX, mouseY, 40, 10); } else if (tool === 3) { // red circles noFill(); stroke(255, 0, 0); ellipse(mouseX, mouseY, 50); } else if (tool === 4) { // green line stroke(0, 150, 0); line(mouseX, mouseY, mouseX, mouseY - random(20, 50)); } else if (tool === 5) { // blue particles stroke(0, 0, 255); point(mouseX + random(-10, 10), mouseY + random(-10, 10)); } else if (tool === 6) { // Grid stroke(200); line(mouseX, 0, mouseX, height); line(0, mouseY, width, mouseY); } else if (tool === 7) { // Eraser stroke(255); strokeWeight(20); line(mouseX, mouseY, pmouseX, pmouseY); strokeWeight(1); } else if (tool === 8) { // color dots fill(random(255), random(255), random(255)); noStroke(); ellipse(mouseX + random(-20, 20), mouseY + random(-20, 20), 10); } else if (tool === 9) { // exclamation point fill(0); textSize(12); text("!", mouseX, mouseY); } else if (tool === 0) { // Reset canvas background(255); } } } function keyPressed() { if (key >= '0' && key <= '9') { tool = int(key); } }
